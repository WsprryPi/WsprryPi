cmake_minimum_required(VERSION 3.20)
project(Singleton LANGUAGES CXX)
include(CTest)
add_library(singleton INTERFACE)
target_include_directories(singleton INTERFACE src)
target_compile_features(singleton INTERFACE cxx_std_20)
if(WIN32)
    target_link_libraries(singleton INTERFACE ws2_32 advapi32)
endif()
if(BUILD_TESTING)
    add_executable(installation_lock_test src/installation_lock_test.cpp)
    target_link_libraries(installation_lock_test PRIVATE singleton)
    if(MINGW)
        target_link_options(installation_lock_test PRIVATE -municode)
    endif()
    if(MSVC)
        target_compile_options(installation_lock_test PRIVATE /W4 /WX /permissive-)
    else()
        target_compile_options(installation_lock_test PRIVATE -Wall -Wextra -Werror)
    endif()
    add_test(NAME installation-lock COMMAND installation_lock_test)
    set_tests_properties(installation-lock PROPERTIES TIMEOUT 20)
    add_executable(singleton_test src/test.cpp)
    target_link_libraries(singleton_test PRIVATE singleton)
    add_test(NAME singleton-udp COMMAND singleton_test)
    set_tests_properties(singleton-udp PROPERTIES TIMEOUT 20)
endif()
