cmake_minimum_required(VERSION 3.20)
project(LCBLog LANGUAGES CXX)

find_package(Threads REQUIRED)
add_library(lcblog src/lcblog.cpp)
target_include_directories(lcblog PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_features(lcblog PUBLIC cxx_std_20)
target_link_libraries(lcblog PUBLIC Threads::Threads)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_path(LCBLOG_SYSTEMD_INCLUDE_DIR systemd/sd-journal.h)
    if(LCBLOG_SYSTEMD_INCLUDE_DIR)
        find_library(LCBLOG_SYSTEMD_LIBRARY systemd REQUIRED)
        target_include_directories(lcblog PRIVATE "${LCBLOG_SYSTEMD_INCLUDE_DIR}")
        target_link_libraries(lcblog PRIVATE "${LCBLOG_SYSTEMD_LIBRARY}")
    endif()
endif()

enable_testing()
add_executable(lcblog_timestamp_test tests/timestamp_test.cpp)
target_link_libraries(lcblog_timestamp_test PRIVATE lcblog)
add_test(NAME lcblog_timestamp COMMAND lcblog_timestamp_test)
set_tests_properties(lcblog_timestamp PROPERTIES TIMEOUT 10)
