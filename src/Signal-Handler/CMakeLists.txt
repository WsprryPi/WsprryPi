cmake_minimum_required(VERSION 3.20)
project(SignalHandler LANGUAGES CXX)
include(CTest)
find_package(Threads REQUIRED)
add_library(signal_handler src/signal_handler.cpp src/signal_wait.cpp src/signal_wait_windows.cpp)
target_include_directories(signal_handler PUBLIC src)
target_compile_features(signal_handler PUBLIC cxx_std_20)
target_link_libraries(signal_handler PUBLIC Threads::Threads)
if(MSVC)
    target_compile_options(signal_handler PRIVATE /W4 /WX /permissive-)
else()
    target_compile_options(signal_handler PRIVATE -Wall -Wextra -Werror)
endif()
if(BUILD_TESTING)
    if(WIN32)
        add_executable(signal_handler_test src/test_windows.cpp)
    else()
        add_executable(signal_handler_test src/test.cpp)
    endif()
    target_link_libraries(signal_handler_test PRIVATE signal_handler)
    if(MINGW)
        target_link_options(signal_handler_test PRIVATE -municode)
    endif()
    add_test(NAME signal-handler-lifecycle COMMAND signal_handler_test)
    set_tests_properties(signal-handler-lifecycle PROPERTIES TIMEOUT 20)
endif()
